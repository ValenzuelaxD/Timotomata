// Programa de monitoreo de sensores
// Esperado: 0 lexicos, 0 sintacticos, 0 semanticos.
sensor voltaje     : ELECTRICO;
sensor temperatura : TERMICO;
sensor presion     : HIDRAULICO;

muestreo = 0.01;

umbral voltaje_max : ELECTRICO = 220;
umbral voltaje_min : ELECTRICO = 100;
umbral temp_max    : TERMICO   = 80;
umbral temp_min    : TERMICO   = -10;
umbral presion_max : HIDRAULICO = 200;

// Reglas de control: gana la primera que se cumple
si voltaje > voltaje_max entonces estado = PICO;
si voltaje < voltaje_min entonces estado = CAIDA;
si temperatura > temp_max entonces estado = INESTABLE;
si temperatura < temp_min entonces estado = CAIDA;
si presion > presion_max entonces estado = NORMAL;

// Regla con abs sobre una senal derivada
calcular(PROMEDIO(CON=voltaje, VENTANA=10), voltaje_prom);
si abs(voltaje_prom) > 50 entonces estado = NORMAL;

sino = NORMAL;
